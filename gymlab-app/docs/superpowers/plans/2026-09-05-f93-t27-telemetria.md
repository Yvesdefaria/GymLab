```ts
// src/domain/telemetry.ts
export interface TelemetryKeys {
  posthogKey?: string
  sentryDsn?: string
}

export type TelemetryState = 'on' | 'off-debug' | 'off-not-configured' | 'off-no-consent'

export interface TelemetryGate {
  prod: boolean
  keys: TelemetryKeys
  consent: boolean
}

// La telemetría solo se activa si la build es de producción, hay claves configuradas
// y el usuario consintió el envío (toggle de Ajustes). Cualquier otra combinación = no-op.
export const shouldEnableTelemetry = ({ prod, keys, consent }: TelemetryGate): boolean =>
  prod && consent && Boolean(keys.posthogKey && keys.sentryDsn)

export const describeTelemetryState = ({ prod, keys, consent }: TelemetryGate): TelemetryState => {
  if (!prod) return 'off-debug'
  if (!keys.posthogKey || !keys.sentryDsn) return 'off-not-configured'
  if (!consent) return 'off-no-consent'
  return 'on'
}
```

```ts
// Telemetría anónima de uso (Sentry + PostHog). Todo el gating vive en domain/telemetry:
// sin prod + claves + consentimiento, estos métodos son no-op y no cargan ningún SDK.
import { shouldEnableTelemetry, describeTelemetryState } from '@/domain/telemetry'

export const TELEMETRY_KEYS = {
  posthogKey: (import.meta.env.VITE_POSTHOG_PROJECT_KEY as string | undefined) || undefined,
  sentryDsn: (import.meta.env.VITE_SENTRY_DSN as string | undefined) || undefined,
} as const

type PosthogLike = {
  capture: (event: string, props?: Record<string, unknown>) => void
  opt_in_capturing: () => void
  opt_out_capturing: () => void
  identify: (id: string) => void
}

let enabled = false
let posthogClient: PosthogLike | null = null
let sentryLoaded = false

const loadPosthog = async (): Promise<PosthogLike | null> => {
  const mod = await import('posthog-js')
  return mod.default as unknown as PosthogLike
}

const gate = (consent: boolean) =>
  shouldEnableTelemetry({
    prod: import.meta.env.PROD,
    keys: TELEMETRY_KEYS,
    consent,
  })

// Se invoca una vez al arrancar la app; idempotente (seguro frente a HMR).
export const initTelemetry = async (consent = true): Promise<void> => {
  if (enabled) return
  if (!gate(consent)) return
  const posthog = await loadPosthog().catch(() => null)
  if (!posthog || !TELEMETRY_KEYS.posthogKey) return
  posthog.init(TELEMETRY_KEYS.posthogKey, {
    api_host: 'https://us.i.posthog.com',
    capture_pageview: false,
    autocapture: {
      css_selector_ignorelist: ['input', 'textarea', 'select', '.ph-no-autocapture', '[data-ph-no-autocapture]'],
    },
    before_send: (event) => {
      const props = event.properties ?? {}
      for (const key of ['text', 'innerText', 'value', 'href', 'name']) delete props[key]
      // Elimina cualquier atributo capturado por autocapture (attr__*): el heatmap solo necesita coordenadas/selector.
      for (const key of Object.keys(props)) if (key.startsWith('attr__')) delete props[key]
      event.properties = props
      return event
    },
  } as Record<string, unknown> as Parameters<PosthogLike['capture']>[0] extends never ? never : object,
  )
  posthogClient = posthog
  if (TELEMETRY_KEYS.sentryDsn) {
    const sentryMod = await import('@sentry/react')
    sentryMod.init({
      dsn: TELEMETRY_KEYS.sentryDsn,
      tracesSampleRate: 0,
      replaysSessionSampleRate: 0,
      replaysOnErrorSampleRate: 0,
    })
    sentryLoaded = true
  }
  enabled = true
}

// Registra un evento de uso. No-op salvo que la telemetría esté activa.
export const track = (event: string, props: Record<string, unknown> = {}): void => {
  if (!enabled || !posthogClient) return
  try {
    posthogClient.capture(event, props)
  } catch {
    /* la telemetría nunca debe romper la app */
  }
}

// Envía un error a Sentry con contexto etiquetado. No-op salvo que esté activa.
export const reportError = (error: unknown, context?: Record<string, unknown>): void => {
  if (!enabled || !sentryLoaded) return
  try {
    const sentry = (() => {
      void import('@sentry/react')
      return undefined
    })()
    void sentry
    void error
    void context
  } catch {
    /* no-op */
  }
}

// Cambio del toggle de Ajustes: ON reactiva (o init), OFF silencia al instante todo envío.
export const applyTelemetryConsent = async (consent: boolean): Promise<void> => {
  if (consent) {
    await initTelemetry(true)
    if (posthogClient) posthogClient.opt_in_capturing()
    return
  }
  enabled = false
  if (posthogClient) posthogClient.opt_out_capturing()
  if (sentryLoaded) {
    try {
      const sentryMod = await import('@sentry/react')
      await sentryMod.close()
    } catch {
      /* no-op */
    }
    sentryLoaded = false
  }
}
```

> **Ojo implementación:** el `posthog.init` tiene tipos estrictos en `posthog-js` v1. El cast del objeto de config (`as Record<string, unknown> as ...`) puede chocar con el tipado del SDK. **Alternativa recomendada:** pasar la config fuertemente tipada y usar las opciones documentadas que sí existen (`autocapture: { css_selector_ignorelist: [...] }`, `before_send`, `capture_pageview: false`, `api_host`). Si `mask_all_inputs`/`mask_text` aceptan el tipado, añadirlos (`posthog.init(key, { ..., mask_all_inputs: true, mask_text: true })`). El **scrub de `before_send` es la garantía**, no los flags.

> **Ojo `reportError`:** el esqueleto de arriba queda sin implementar de propósito por simetría de no-op en tests. Implementación real:
>   - importar `captureException` de `@sentry/react` en un módulo cargado solo cuando `sentryLoaded` es true, vía un acceso dinámico cacheado, o simplemente `const { captureException } = await import('@sentry/react')` en el cuerpo de `reportError` tras el guard, y llamar `captureException(error, { contexts: { telemetry: context ?? {} } })`. Ajustar para que `tsc -b` pase y que seguir importando `@sentry/react` en Safari de Vite dev sea inofensivo.
>   - Ambas opacity: NO escribir el import de `@sentry/react` a nivel de topo del módulo si rompe el tree-shake; el `await import()` dentro de la función es el camino limpio.

```ts
import { useCallback } from 'react'
import { useSettings } from '@/hooks/useSettings'
import { applyTelemetryConsent } from '@/lib/telemetry'

export const useTelemetryConsent = () => {
  const { settings, update } = useSettings()
  const consent = settings.telemetry

  const setConsent = useCallback(
    async (value: boolean) => {
      await update({ telemetry: value })
      void applyTelemetryConsent(value)
    },
    [update]
  )

  return { consent, setConsent }
}
```

## Notas de implementación (leer antes de empezar)

- **No romper el tree-shake:** `@sentry/react` y `posthog-js` solo se importan con `await import(...)` dentro de funciones guardadas por gating. Nunca al topo del módulo.
- **Doble init/HMR:** `initTelemetry` es idempotente (`if (enabled) return`).
- **Consentimiento runtime:** OFF → `opt_out_capturing()` + `Sentry.close()`; ON → `initTelemetry()` + `opt_in_capturing()`. No hace falta recargar la página.
- **Tests no se ven afectados:** vitest corre con `import.meta.env.PROD = false` → `shouldEnableTelemetry` es false → todo no-op. El E2E corre en dev sin claves → 0 peticiones.
- **PostHog endpoint:** `https://us.i.posthog.com` (por defecto en el SDK); si el usuario crea el proyecto en la región UE, cambia a `https://eu.i.posthog.com` — configurar en la Task 2 según la región elegida por el usuario.
- **`legal.test.ts`:** si compara estructura ES/EN (mismo nº de párrafos), mantener los arrays de `cuerpo` de la sección analítica con idéntica longitud en ambos idiomas (Task 6).
